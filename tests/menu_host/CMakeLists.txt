cmake_minimum_required(VERSION 3.16)
project(kim_menu_probe C CXX)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH ${CMAKE_CURRENT_LIST_DIR}/lv_conf.h CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../firmware/managed_components/lvgl__lvgl lvgl)
add_executable(menu_probe menu_probe.c)
target_compile_options(menu_probe PRIVATE -UNDEBUG)
target_link_libraries(menu_probe PRIVATE lvgl m)
add_executable(avatar_render_probe avatar_render_probe.c)
target_compile_options(avatar_render_probe PRIVATE -UNDEBUG)
target_link_libraries(avatar_render_probe PRIVATE lvgl m)
add_executable(avatar_sequence_probe avatar_sequence_probe.c ../../firmware/main/avatar_player.c)
target_compile_definitions(avatar_sequence_probe PRIVATE KIM_AVATAR_HOST_TEST)
target_include_directories(avatar_sequence_probe PRIVATE ../../firmware/main ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(avatar_sequence_probe PRIVATE -UNDEBUG)
target_link_libraries(avatar_sequence_probe PRIVATE lvgl m)
